cmake_minimum_required(VERSION 3.25)
project(XunTradeOMSEMS VERSION 0.1.0 LANGUAGES CXX)

# One language policy for the library, CPython module and every native test.
if(DEFINED CMAKE_CXX_STANDARD AND NOT CMAKE_CXX_STANDARD STREQUAL "20")
  message(FATAL_ERROR "XunTradeOMSEMS requires CMAKE_CXX_STANDARD=20")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  # MSVC otherwise reports the historical __cplusplus value even in C++20 mode.
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/Zc:__cplusplus>")
endif()

find_package(Threads REQUIRED)
add_library(xuntrade_kernels STATIC native/src/kernels.cpp)
target_include_directories(xuntrade_kernels PUBLIC native/include)
target_compile_features(xuntrade_kernels PUBLIC cxx_std_20)
set_target_properties(xuntrade_kernels PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
target_link_libraries(xuntrade_kernels PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(xuntrade_kernels PRIVATE /W4 /EHsc)
else()
  target_compile_options(xuntrade_kernels PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(XUNTRADE_BUILD_PYTHON "Build CPython extension with CMake (pip builds it separately)" OFF)
if(XUNTRADE_BUILD_PYTHON)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter Development.Module)
  Python3_add_library(_native MODULE WITH_SOABI native/src/python_module.cpp)
  target_link_libraries(_native PRIVATE xuntrade_kernels)
  install(TARGETS _native LIBRARY DESTINATION xuntrade_omsems RUNTIME DESTINATION xuntrade_omsems)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(xuntrade_kernel_tests native/tests/test_kernels.cpp)
  target_link_libraries(xuntrade_kernel_tests PRIVATE xuntrade_kernels)
  add_test(NAME xuntrade_kernel_tests COMMAND xuntrade_kernel_tests)
endif()
